// NEW: Top Buy alert (A3 verified — 사용자 약점 L5 처방)

// Swing high (좌우 5캔들 fractal)
swing_high = ta.pivothigh(high, 5, 5)
var float last_swing_high = na
var float last_swing_low  = na
var int   last_sh_idx = na

if not na(swing_high)
    last_swing_high := high[5]
    last_sh_idx := bar_index - 5

swing_low  = ta.pivotlow(low, 5, 5)
if not na(swing_low)
    last_swing_low := low[5]

// Fib retrace from current price to last swing
leg_size     = not na(last_swing_high) and not na(last_swing_low) ?
                last_swing_high - last_swing_low : 0.0
retrace_pct  = leg_size > 0 ? (last_swing_high - close) / leg_size : 0.0
bars_since_sh = bar_index - last_sh_idx

// Top Buy 위험 = swing high 직후 5봉 안 OR fib retrace < 23.6%
top_buy_risk = bars_since_sh < 5 or
               (retrace_pct >= 0 and retrace_pct < 0.236)

// Visual: 빨간 ▲ (LONG 진입 시 위험 알림)
plotshape(top_buy_risk and ema_up, title="🚨 Top Buy Risk",
          style=shape.labeldown, location=location.abovebar,
          color=color.new(color.red, 20), size=size.tiny,
          text="TOP", textcolor=color.white)

// Alert
alertcondition(top_buy_risk and ema_up,
   title="Top Buy Risk Alert",
   message="⚠️ Top Buy Risk: swing high < 5b ago OR fib retrace < 23.6% — entry 금지")
